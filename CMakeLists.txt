INCLUDE(CheckIncludeFile)

FIND_PACKAGE(CURL REQUIRED)
FIND_PACKAGE(LibArchive REQUIRED)

SET(BANQUISE_AGENT_CRYPTO_BACKEND "AUTO" CACHE STRING
  "Crypto backend: AUTO, OPENSSL, or WOLFSSL")
SET_PROPERTY(CACHE BANQUISE_AGENT_CRYPTO_BACKEND PROPERTY STRINGS AUTO OPENSSL WOLFSSL)

IF(BANQUISE_AGENT_CRYPTO_BACKEND STREQUAL "AUTO")
  SET(BANQUISE_AGENT_USE_WOLFSSL OFF)
  IF(EXISTS "${CMAKE_BINARY_DIR}/include/my_config.h")
    FILE(READ "${CMAKE_BINARY_DIR}/include/my_config.h" BANQUISE_MY_CONFIG)
    IF(BANQUISE_MY_CONFIG MATCHES "#define[ \t]+HAVE_WOLFSSL([ \t]|$)")
      SET(BANQUISE_AGENT_USE_WOLFSSL ON)
    ENDIF()
  ENDIF()
  IF(DEFINED SSL_DEFINES AND SSL_DEFINES MATCHES "HAVE_WOLFSSL")
    SET(BANQUISE_AGENT_USE_WOLFSSL ON)
  ENDIF()
ELSEIF(BANQUISE_AGENT_CRYPTO_BACKEND STREQUAL "WOLFSSL")
  SET(BANQUISE_AGENT_USE_WOLFSSL ON)
ELSEIF(BANQUISE_AGENT_CRYPTO_BACKEND STREQUAL "OPENSSL")
  SET(BANQUISE_AGENT_USE_WOLFSSL OFF)
ELSE()
  MESSAGE(FATAL_ERROR "BANQUISE_AGENT_CRYPTO_BACKEND must be AUTO, OPENSSL, or WOLFSSL")
ENDIF()

IF(BANQUISE_AGENT_USE_WOLFSSL)
  # Keep bundled wolfSSL headers in the same configured mode as MariaDB.
  ADD_DEFINITIONS(${SSL_DEFINES})
  IF(EXISTS "${CMAKE_BINARY_DIR}/extra/wolfssl/user_settings.h")
    INCLUDE_DIRECTORIES("${CMAKE_BINARY_DIR}/extra/wolfssl")
  ENDIF()
  SET(BANQUISE_AGENT_CRYPTO_SOURCES banquise_crypto_wolfssl.cc)
  IF(EXISTS "${CMAKE_SOURCE_DIR}/extra/wolfssl/wolfssl/wolfcrypt/src/blake2b.c")
    ADD_DEFINITIONS(-DHAVE_BLAKE2 -DHAVE_BLAKE2B)
    LIST(APPEND BANQUISE_AGENT_CRYPTO_SOURCES
      "${CMAKE_SOURCE_DIR}/extra/wolfssl/wolfssl/wolfcrypt/src/blake2b.c")
  ENDIF()
  IF(TARGET wolfssl)
    SET(BANQUISE_AGENT_CRYPTO_LIBRARY wolfssl)
  ELSEIF(SSL_LIBRARIES)
    SET(BANQUISE_AGENT_CRYPTO_LIBRARY ${SSL_LIBRARIES})
  ELSE()
    MESSAGE(FATAL_ERROR "WolfSSL selected but no MariaDB wolfSSL library is available")
  ENDIF()
ELSE()
  FIND_PACKAGE(OpenSSL REQUIRED COMPONENTS Crypto)
  SET(BANQUISE_AGENT_CRYPTO_SOURCES banquise_crypto_openssl.cc)
  SET(BANQUISE_AGENT_CRYPTO_LIBRARY OpenSSL::Crypto)
ENDIF()

MYSQL_ADD_PLUGIN(BANQUISE_AGENT banquise_agent.cc ${BANQUISE_AGENT_CRYPTO_SOURCES}
  MODULE_ONLY
  RECOMPILE_FOR_EMBEDDED
  LINK_LIBRARIES CURL::libcurl LibArchive::LibArchive ${BANQUISE_AGENT_CRYPTO_LIBRARY})
